import { Request, Response, NextFunction } from 'express';

export class HttpError extends Error {
  constructor(public status: number, message: string) { super(message); }
}
type H = (req: Request, res: Response, next: NextFunction) => unknown;
export const wrap = (fn: H) => (req: Request, res: Response, next: NextFunction) =>
  Promise.resolve(fn(req, res, next)).catch(next);

export function notFound(_req: Request, res: Response) { res.status(404).json({ error: 'المسار غير موجود' }); }
export function errorHandler(err: any, _req: Request, res: Response, _next: NextFunction) {
  let status = err instanceof HttpError ? err.status : 500;
  // ملف مرفوع أكبر من الحد المسموح: رسالة واضحة بحالة 400 بدل 500
  if (err?.code === 'LIMIT_FILE_SIZE') { status = 400; err.message = 'حجم الملف أكبر من الحد المسموح'; }
  if (status >= 500) console.error(err);
  res.status(status).json({ error: err.message || 'حدث خطأ في الخادم' });
}

import bcrypt from 'bcryptjs';
import jwt from 'jsonwebtoken';
import { env } from '../config';
import { q } from '../database/pool';
import { HttpError } from '../middleware/errors';

// حد بسيط ضد التخمين: 5 محاولات فاشلة → قفل 15 دقيقة لنفس اسم المستخدم
const fails = new Map<string, { n: number; until: number }>();

export const hashPassword = (plain: string) => bcrypt.hash(plain, 10);
export const verifyPassword = (plain: string, hash: string) => bcrypt.compare(plain, hash);

export async function login(username: string, password: string) {
  const key = String(username ?? '').trim().toLowerCase();
  const f = fails.get(key);
  // القفل فعّال طوال مدته بغض النظر عن العدّاد
  if (f && f.until > Date.now())
    throw new HttpError(429, 'تم تجاوز عدد المحاولات المسموح، حاولي بعد قليل');
  const { rows } = await q<any>(
    'SELECT id, username, password_hash, role, active, token_version, student_id FROM users WHERE lower(username) = $1',
    [key]);
  const u = rows[0];
  if (!u || !(await verifyPassword(String(password ?? ''), u.password_hash))) {
    const c = fails.get(key) ?? { n: 0, until: 0 };
    c.n++;
    if (c.n >= 5) { c.until = Date.now() + 15 * 60 * 1000; c.n = 0; }
    fails.set(key, c);
    throw new HttpError(401, 'اسم المستخدم أو كلمة المرور غير صحيحة');
  }
  if (!u.active) throw new HttpError(403, 'هذا الحساب موقوف، يرجى مراجعة المعلمة');
  fails.delete(key);
  const token = jwt.sign({ uid: u.id, tv: u.token_version }, env.jwtSecret, { expiresIn: '8h' });
  return { token, user: { id: u.id, username: u.username, role: u.role, student_id: u.student_id } };
}

export async function changePassword(userId: number, newPassword: string, current?: string, requireCurrent = false) {
  const { rows } = await q<{ password_hash: string }>('SELECT password_hash FROM users WHERE id = $1', [userId]);
  if (!rows[0]) throw new HttpError(404, 'المستخدم غير موجود');
  if (requireCurrent) {
    if (!current) throw new HttpError(400, 'أدخلي كلمة المرور الحالية');
    if (!(await verifyPassword(current, rows[0].password_hash))) throw new HttpError(400, 'كلمة المرور الحالية غير صحيحة');
  }
  if (!newPassword || String(newPassword).length < 6) throw new HttpError(400, 'كلمة المرور يجب ألا تقل عن 6 أحرف');
  await q('UPDATE users SET password_hash = $1, token_version = token_version + 1 WHERE id = $2',
    [await hashPassword(newPassword), userId]);
}

/** استيراد جماعي: يتجاهل المكرر (اسم + صف + شعبة) ويرجع الإحصاء */
export async function importRows(rowsIn: any[]) {
  if (!Array.isArray(rowsIn) || !rowsIn.length) throw new HttpError(400, 'لا توجد صفوف للاستيراد');
  let added = 0, skipped = 0;
  for (const r of rowsIn.slice(0, 1000)) {
    const name = String(r.name ?? '').trim();
    const grade = String(r.grade ?? '').trim();
    const section = String(r.section ?? '').trim();
    if (!name || !grade || !section) { skipped++; continue; }
    const ex = await q('SELECT 1 FROM students WHERE name = $1 AND grade = $2 AND section = $3', [name, grade, section]);
    if (ex.rowCount) { skipped++; continue; }
    const ins = await q(
      "INSERT INTO students(code, name, grade, section) VALUES ('TMP-' || md5(random()::text), $1,$2,$3) RETURNING id",
      [name, grade, section]);
    await q("UPDATE students SET code = 'ST-' || lpad(id::text, 3, '0') WHERE id = $1", [ins.rows[0].id]);
    added++;
  }
  return { added, skipped };
}

import { q } from '../database/pool';
import { HttpError } from '../middleware/errors';
import { AuthUser, assertStudentAccess } from '../middleware/auth';

const isDate = (s: any) => typeof s === 'string' && /^\d{4}-\d{2}-\d{2}$/.test(s);
const score = (v: any) => {
  if (v === undefined || v === null || v === '') return null;
  const n = Number(v);
  if (!Number.isInteger(n) || n < 1 || n > 5) throw new HttpError(400, 'الدرجة يجب أن تكون بين 1 و 5');
  return n;
};

export async function list(user: AuthUser, f: { student_id?: string; from?: string; to?: string }) {
  let sid: number;
  if (user.role === 'student') {
    // الطالبة التي تطلب صراحةً بيانات طالبة أخرى تُرفض 403 — لا تجاوز صامت
    if (f.student_id !== undefined && f.student_id !== '' && Number(f.student_id) !== user.studentId)
      throw new HttpError(403, 'غير مصرح بالوصول لهذه البيانات');
    sid = user.studentId!;
  } else {
    sid = Number(f.student_id);
  }
  if (!sid) throw new HttpError(400, 'حددي طالبة');
  assertStudentAccess(user, sid);
  const { rows } = await q(
    `SELECT * FROM evaluations WHERE student_id = $1
       AND ($2::date IS NULL OR eval_date >= $2) AND ($3::date IS NULL OR eval_date <= $3)
     ORDER BY eval_date DESC, id DESC`,
    [sid, f.from || null, f.to || null]);
  return rows;
}

export async function create(user: AuthUser, b: any) {
  const sid = Number(b.student_id);
  if (!sid) throw new HttpError(400, 'حددي الطالبة');
  if (!isDate(b.eval_date)) throw new HttpError(400, 'التاريخ غير صالح');
  const attendance = b.attendance;
  if (!['present', 'excused', 'absent'].includes(attendance)) throw new HttpError(400, 'حالة الحضور غير صالحة');
  const st = await q('SELECT 1 FROM students WHERE id = $1', [sid]);
  if (!st.rowCount) throw new HttpError(404, 'الطالبة غير موجودة');
  let behavior: number | null = null, participation: number | null = null, homework: number | null = null;
  if (attendance === 'present') {
    behavior = score(b.behavior); participation = score(b.participation); homework = score(b.homework);
    if (behavior === null || participation === null || homework === null)
      throw new HttpError(400, 'أكملي درجات السلوك والمشاركة والواجب');
  } else if (b.behavior != null || b.participation != null || b.homework != null) {
    throw new HttpError(400, 'لا تُسجَّل درجات لطالبة غير حاضرة'); // الحضور مستقل عن الدرجات
  }
  const { rows } = await q(
    `INSERT INTO evaluations(student_id, eval_date, attendance, behavior, participation, homework, notes, created_by)
     VALUES ($1,$2,$3,$4,$5,$6,$7,$8) RETURNING *`,
    [sid, b.eval_date, attendance, behavior, participation, homework, b.notes || null, user.id]);
  return rows[0];
}

export async function remove(id: number) {
  const { rows } = await q('DELETE FROM evaluations WHERE id = $1 RETURNING id, student_id', [id]);
  if (!rows[0]) throw new HttpError(404, 'التقييم غير موجود');
  return rows[0];
}

import { q } from '../database/pool';

export async function list(user: AuthUser, f: { grade?: string; section?: string }) {
  if (user.role === 'student') {
    const sid = user.studentId!;
    const { rows } = await q(
      `SELECT w.id, w.title, w.description, w.grade, w.section, w.work_date,
              COALESCE(t.done, false) AS done,
              (SELECT id FROM files fi WHERE fi.worksheet_id = w.id LIMIT 1) AS file_id
       FROM worksheets w
       LEFT JOIN worksheet_targets t ON t.worksheet_id = w.id AND t.student_id = $1
       WHERE w.grade = (SELECT grade FROM students WHERE id = $1)
         AND w.section = (SELECT section FROM students WHERE id = $1)
         AND (NOT EXISTS (SELECT 1 FROM worksheet_targets x WHERE x.worksheet_id = w.id)
              OR t.student_id IS NOT NULL)
       ORDER BY w.work_date DESC, w.id DESC`, [sid]);
    return rows;
  }
  const { rows } = await q(
    `SELECT w.*,
       COALESCE(json_agg(json_build_object('student_id', s.id, 'name', s.name, 'done', t.done) ORDER BY s.name)
                FILTER (WHERE s.id IS NOT NULL), '[]') AS targets
     FROM worksheets w
     LEFT JOIN worksheet_targets t ON t.worksheet_id = w.id
     LEFT JOIN students s ON s.id = t.student_id
     WHERE ($1::text IS NULL OR w.grade = $1) AND ($2::text IS NULL OR w.section = $2)
     GROUP BY w.id ORDER BY w.work_date DESC, w.id DESC`,
    [f.grade || null, f.section || null]);
  return rows;
}

export async function create(user: AuthUser, b: any, file?: UpFile) {
  if (!b.title || !String(b.title).trim()) throw new HttpError(400, 'عنوان ورقة العمل مطلوب');
  if (!b.grade || !b.section) throw new HttpError(400, 'الصف والشعبة مطلوبان');
  const { rows } = await q(
    `INSERT INTO worksheets(title, description, grade, section, work_date, created_by)
     VALUES ($1,$2,$3,$4, COALESCE($5::date, CURRENT_DATE), $6) RETURNING *`,
    [String(b.title).trim(), b.description || null, b.grade, b.section, b.work_date || null, user.id]);
  const w = rows[0];
  const ids: number[] = Array.isArray(b.ids) ? b.ids.map(Number).filter(Boolean) : [];
  if (ids.length) {
    await q(
      'INSERT INTO worksheet_targets(worksheet_id, student_id) SELECT $1, id FROM students WHERE id = ANY($2::int[]) ON CONFLICT DO NOTHING',
      [w.id, ids]);
  }
  if (file) await filesSvc.register({
    stored_name: file.filename, original_name: file.originalname, mime: file.mimetype,
    size: file.size, kind: 'worksheet', student_id: null, worksheet_id: w.id, achievement_id: null, uploaded_by: user.id
  });
  return w;
}

     RETURNING *`, [worksheetId, studentId, done]);
  return rows[0];
}

import { q, pool } from '../database/pool';
import { HttpError } from '../middleware/errors';

const norm = (r: any) => ({ ...r, score: r.score == null ? null : Number(r.score), max_score: Number(r.max_score) });

export async function list(user: AuthUser, f: { kind?: string; grade?: string; section?: string }) {
  if (user.role === 'student') {
    const sid = user.studentId!;
    const { rows } = await q(
      `SELECT t.id, t.title, t.kind, t.max_score, t.test_date, r.score, r.note
       FROM tests t
       LEFT JOIN results r ON r.test_id = t.id AND r.student_id = $1
       WHERE t.grade = (SELECT grade FROM students WHERE id = $1)
         AND t.section = (SELECT section FROM students WHERE id = $1)
         AND ($2::text IS NULL OR t.kind = $2)
       ORDER BY t.test_date DESC, t.id DESC`, [sid, f.kind || null]);
    return rows.map(norm);
  }
  const { rows } = await q(
    `SELECT t.*, (SELECT count(*) FROM results r WHERE r.test_id = t.id) AS results_count
     FROM tests t
     WHERE ($1::text IS NULL OR t.kind = $1) AND ($2::text IS NULL OR t.grade = $2) AND ($3::text IS NULL OR t.section = $3)
     ORDER BY t.test_date DESC, t.id DESC`, [f.kind || null, f.grade || null, f.section || null]);
  return rows.map(norm);
}

export async function create(user: AuthUser, b: any) {
  if (!b.title || !String(b.title).trim()) throw new HttpError(400, 'عنوان الاختبار مطلوب');
  if (!['test', 'nafis'].includes(b.kind)) throw new HttpError(400, 'نوع الاختبار غير صالح');
  const max = Number(b.max_score);
  if (!Number.isFinite(max) || max <= 0) throw new HttpError(400, 'الدرجة العظمى غير صالحة');
  if (!b.grade || !b.section) throw new HttpError(400, 'الصف والشعبة مطلوبان');
  const { rows } = await q(
    `INSERT INTO tests(title, kind, max_score, test_date, grade, section, created_by)
     VALUES ($1,$2,$3, COALESCE($4::date, CURRENT_DATE), $5,$6,$7) RETURNING *`,
    [String(b.title).trim(), b.kind, max, b.test_date || null, b.grade, b.section, user.id]);
  return norm(rows[0]);
}

export async function saveResults(user: AuthUser, testId: number, rowsIn: any[]) {
  const t = await q('SELECT * FROM tests WHERE id = $1', [testId]);
  if (!t.rows[0]) throw new HttpError(404, 'الاختبار غير موجود');
  const max = Number(t.rows[0].max_score);
  const client = await pool.connect();
  try {
    await client.query('BEGIN');
    for (const r of rowsIn ?? []) {
      const sid = Number(r.student_id), sc = Number(r.score);
      if (!sid || !Number.isFinite(sc) || sc < 0 || sc > max)
        throw new HttpError(400, 'درجة غير صالحة (تأكدي أنها بين 0 والدرجة العظمى)');
      await client.query(
        `INSERT INTO results(test_id, student_id, score, note) VALUES ($1,$2,$3,$4)
         ON CONFLICT (test_id, student_id) DO UPDATE SET score = EXCLUDED.score, note = EXCLUDED.note, updated_at = now()`,
        [testId, sid, sc, r.note || null]);
    }
    await client.query('COMMIT');
  } catch (e) { await client.query('ROLLBACK'); throw e; }
  finally { client.release(); }
  return { saved: (rowsIn ?? []).length, test_id: testId, updated_by: user.id };
}

export async function remove(id: number) {
  const { rows } = await q('DELETE FROM tests WHERE id = $1 RETURNING id', [id]);
  if (!rows[0]) throw new HttpError(404, 'الاختبار غير موجود');
  return rows[0];
}

export async function list(user: AuthUser, studentId?: number) {
  if (user.role === 'student') {
    const sid = user.studentId!;
    const { rows } = await q(
      `SELECT sk.id, sk.name, sk.description, l.level, l.note
       FROM skills sk LEFT JOIN skill_levels l ON l.skill_id = sk.id AND l.student_id = $1
       ORDER BY sk.name`, [sid]);
    return rows;
  }
  const sid = studentId ?? null;
  assertStudentAccess(user, sid ?? -1);
  const { rows } = await q(
    `SELECT sk.*, COALESCE(json_agg(json_build_object('student_id', l.student_id, 'level', l.level, 'note', l.note))
            FILTER (WHERE l.student_id IS NOT NULL), '[]') AS levels
     FROM skills sk LEFT JOIN skill_levels l ON l.skill_id = sk.id
       AND ($1::int IS NULL OR l.student_id = $1)
     GROUP BY sk.id ORDER BY sk.name`, [sid]);
  return rows;
}

export async function create(user: AuthUser, b: any) {
  if (!b.name || !String(b.name).trim()) throw new HttpError(400, 'اسم المهارة مطلوب');
  try {
    const { rows } = await q('INSERT INTO skills(name, description) VALUES ($1,$2) RETURNING *',
      [String(b.name).trim(), b.description || null]);
    return rows[0];
  } catch (e: any) {
    if (e.code === '23505') throw new HttpError(409, 'هذه المهارة موجودة مسبقاً');
    throw e;
  }
}

export async function setLevels(user: AuthUser, skillId: number, rowsIn: any[]) {
  const sk = await q('SELECT 1 FROM skills WHERE id = $1', [skillId]);
  if (!sk.rowCount) throw new HttpError(404, 'المهارة غير موجودة');
  const client = await pool.connect();
  try {
    await client.query('BEGIN');
    for (const r of rowsIn ?? []) {
      const sid = Number(r.student_id), lv = Number(r.level);
      if (!sid || !Number.isInteger(lv) || lv < 1 || lv > 5)
        throw new HttpError(400, 'مستوى الإتقان يجب أن يكون بين 1 و 5');
      await client.query(
        `INSERT INTO skill_levels(skill_id, student_id, level, note) VALUES ($1,$2,$3,$4)
         ON CONFLICT (skill_id, student_id) DO UPDATE SET level = EXCLUDED.level, note = EXCLUDED.note, updated_at = now()`,
        [skillId, sid, lv, r.note || null]);
    }
    await client.query('COMMIT');
  } catch (e) { await client.query('ROLLBACK'); throw e; }
  finally { client.release(); }
  return { saved: (rowsIn ?? []).length, skill_id: skillId, updated_by: user.id };
}

export async function remove(id: number) {
  const { rows } = await q('DELETE FROM skills WHERE id = $1 RETURNING id', [id]);
  if (!rows[0]) throw new HttpError(404, 'المهارة غير موجودة');
  return rows[0];
}

    'SELECT id, sender, body, sent_at FROM messages WHERE student_id = $1 ORDER BY sent_at ASC LIMIT 200', [studentId]);

export function sundayOf(dstr: string): string {
  const d = new Date(dstr + 'T00:00:00Z');

  d.setUTCDate(d.getUTCDate() - d.getUTCDay()); // بداية الأسبوع = الأحد
  return d.toISOString().slice(0, 10);
}

export function accessible(user: AuthUser, f: any): boolean {

  return false;
}

/** اتجاه الطالبة: لا يُصدَر حكم قبل 4 تقييمات حاضرة على الأقل */
function trend(series: { d: string; v: number }[]) {
  if (series.length < 4) return { enough: false, n: series.length, slope: null };
  const xs = series.map(e => Date.parse(e.d) / 86400000), ys = series.map(e => e.v);
  const n = xs.length, mx = xs.reduce((a, b) => a + b) / n, my = ys.reduce((a, b) => a + b) / n;
  let a1 = 0, b1 = 0;
  for (let i = 0; i < n; i++) { a1 += (xs[i] - mx) * (ys[i] - my); b1 += (xs[i] - mx) ** 2; }
  return { enough: true, n, slope: b1 ? +(a1 / b1).toFixed(4) : 0 };
}

export async function summary(f: { from?: string; to?: string; grade?: string; section?: string }) {
  const { rows } = await q(
    `SELECT s.id, s.code, s.name, s.grade, s.section,
       count(e.id) FILTER (WHERE e.attendance = 'present')  AS present,
       count(e.id) FILTER (WHERE e.attendance = 'excused')  AS excused,
       count(e.id) FILTER (WHERE e.attendance = 'absent')   AS absent,
       round(avg(e.behavior)      FILTER (WHERE e.attendance = 'present')::numeric, 2)::float8 AS behavior,
       round(avg(e.participation) FILTER (WHERE e.attendance = 'present')::numeric, 2)::float8 AS participation,
       round(avg(e.homework)      FILTER (WHERE e.attendance = 'present')::numeric, 2)::float8 AS homework,
       max(e.eval_date) AS last_eval
     FROM students s
     LEFT JOIN evaluations e ON e.student_id = s.id
       AND ($1::date IS NULL OR e.eval_date >= $1) AND ($2::date IS NULL OR e.eval_date <= $2)
     WHERE s.active AND ($3::text IS NULL OR s.grade = $3) AND ($4::text IS NULL OR s.section = $4)
     GROUP BY s.id ORDER BY s.grade, s.section, s.name`,
    [f.from || null, f.to || null, f.grade || null, f.section || null]);

  const { rows: series } = await q(
    `SELECT student_id, eval_date AS d,
       ((COALESCE(behavior,0) + COALESCE(participation,0) + COALESCE(homework,0))
        / NULLIF((behavior IS NOT NULL)::int + (participation IS NOT NULL)::int + (homework IS NOT NULL)::int, 0))::float8 AS v
     FROM evaluations
     WHERE attendance = 'present' AND ($1::date IS NULL OR eval_date >= $1) AND ($2::date IS NULL OR eval_date <= $2)
     ORDER BY student_id, eval_date`, [f.from || null, f.to || null]);

  const byS = new Map<number, { d: string; v: number }[]>();
  for (const r of series) {
    if (!byS.has(r.student_id)) byS.set(r.student_id, []);
    byS.get(r.student_id)!.push({ d: r.d, v: r.v });
  }
  return rows.map(r => ({ ...r, trend: trend(byS.get(r.id) ?? []) }));
}

export async function compare(f: any) {
  const [a, b] = await Promise.all([
    summary({ from: f.fromA, to: f.toA, grade: f.grade, section: f.section }),
    summary({ from: f.fromB, to: f.toB, grade: f.grade, section: f.section })
  ]);
  const mapB = new Map(b.map((r: any) => [r.id, r]));
  return a.map((ra: any) => {
    const rb: any = mapB.get(ra.id);
    const delta = (k: string) => (rb && ra[k] != null && rb[k] != null) ? +(rb[k] - ra[k]).toFixed(2) : null;
    return {
      student: { id: ra.id, name: ra.name, grade: ra.grade, section: ra.section },
      periodA: ra, periodB: rb ?? null,
      delta: { behavior: delta('behavior'), participation: delta('participation'), homework: delta('homework') },
      trendEnough: ra.trend.enough && (!rb || rb.trend.enough) // لا مقارنة اتجاه بدون بيانات كافية
    };
  });
}

  if (role === 'student') {
    studentId = Number(b.student_id);
    if (!studentId) throw new HttpError(400, 'يجب ربط الحساب بالطالبة عبر رمزها الثابت');
    const s = await q('SELECT id FROM students WHERE id = $1', [studentId]);
    if (!s.rowCount) throw new HttpError(404, 'الطالبة غير موجودة');
    const taken = await q('SELECT 1 FROM users WHERE student_id = $1', [studentId]);
    if (taken.rowCount) throw new HttpError(409, 'لهذه الطالبة حساب بالفعل');
  }
  try {
    const { rows } = await q(
      'INSERT INTO users(username, password_hash, role, student_id) VALUES ($1,$2,$3,$4) RETURNING id, username, role, active, student_id',
      [username, await hashPassword(String(b.password)), role, studentId]);
    return rows[0];
  } catch (e: any) {
    if (e.code === '23505') throw new HttpError(409, 'اسم المستخدم مستخدم مسبقاً');
    throw e;
  }
}

export async function update(actor: AuthUser, id: number, b: any) {
  const { rows } = await q('SELECT * FROM users WHERE id = $1', [id]);
  const u = rows[0];
  if (!u) throw new HttpError(404, 'المستخدم غير موجود');
  if (actor.role !== 'admin' && u.role !== 'student') throw new HttpError(403, 'غير مصرح');
  // student_id غير قابل للتعديل عمداً: الربط بالرمز الثابت لا يتغير
  try {
    if (b.username) {
      const un = String(b.username).trim().toLowerCase();
      await q('UPDATE users SET username = $2 WHERE id = $1', [id, un]);
    }
    if (b.active !== undefined) await q('UPDATE users SET active = $2 WHERE id = $1', [id, !!b.active]);
    if (b.password) {
      if (String(b.password).length < 6) throw new HttpError(400, 'كلمة المرور قصيرة جداً');
      await q('UPDATE users SET password_hash = $2, token_version = token_version + 1 WHERE id = $1',
        [id, await hashPassword(String(b.password))]);
    }
  } catch (e: any) {
    if (e.code === '23505') throw new HttpError(409, 'اسم المستخدم مستخدم مسبقاً');
    throw e;
  }
  const { rows: out } = await q('SELECT id, username, role, active, student_id FROM users WHERE id = $1', [id]);
  return out[0];
}

export async function remove(id: number) {
  const { rows } = await q('DELETE FROM users WHERE id = $1 RETURNING id, username', [id]);
  if (!rows[0]) throw new HttpError(404, 'المستخدم غير موجود');
  return rows[0];
}

import path from 'path';
import fs from 'fs';
import { wrap } from '../middleware/errors';
import * as svc from '../services/files';
import { audit } from '../services/audit';
import { env } from '../config';
import { HttpError } from '../middleware/errors';

export const download = wrap(async (req, res) => {
  const f = await svc.getForDownload(+req.params.id);
  // 404 بدل 403 عند عدم الملكية: لا نكشف حتى وجود الملف لمن يحاول تخمين الروابط
  if (!f || !svc.accessible(req.user!, f)) throw new HttpError(404, 'الملف غير موجود');
  const abs = path.join(env.uploadDirAbs, f.stored_name);
  if (!fs.existsSync(abs)) throw new HttpError(404, 'الملف غير موجود');
  await audit(req, 'file.download', 'file', f.id, { kind: f.kind });
  res.download(abs, f.original_name);
});
