# منصة تقييم طالبات مادة لغتي — نسخة Full-Stack

تطبيق حقيقي ثلاثي الطبقات: واجهة React + TypeScript، خادم Node.js/Express + TypeScript،
قاعدة PostgreSQL. جميع الصلاحيات تُفرض على الخادم، ولا توجد أي بيانات أو مفاتيح سرية في كود الواجهة.

## المتطلبات
- Node.js 20 أو أحدث
- PostgreSQL 14 أو أحدث (على Windows: مثبّت من postgresql.org يكفي، والتشغيل من pgAdmin أو الخدمة التلقائية)

## التشغيل المحلي (من المرة الأولى)
1. انسخي `.env.example` إلى `.env` وعدّلي `DATABASE_URL` و`JWT_SECRET`.
2. إنشاء قاعدة البيانات والجداول والحساب الأول:
   ```
   cd backend
   npm install
   npm run db:create     # ينشئ قاعدة lughati إن لم توجد
   npm run migrate       # ينشئ الجداول والفهارس
   npm run seed          # الإعدادات الافتراضية + حساب المعلمة admin
   ```
3. نقل أسماء الطالبات (إحدى الطريقتين):
   - الصقي الأسماء في `database/seeds/students.csv` (سطر: الاسم،الصف،الشعبة) ثم:
     `npm run import:students -- ../database/seeds/students.csv`
   - أو من الواجهة: الطالبات ← «استيراد سريع» (الصقي مباشرة من تصدير CSV في النسخة السابقة).
4. تشغيل الخادم: `npm run dev`  (منفذ 4000)
5. تشغيل الواجهة (نافذة ثانية):
   ```
   cd ../frontend
   npm install
   npm run dev          # منفذ 5173 — افتحي http://localhost:5173
   ```
6. الدخول الأول: `admin` / `lughati1234` ← **غيّري كلمة المرور فوراً من الإعدادات**.

## الاختبارات (تكامل حقيقية ضد قاعدة فعلية)
1. أنشئي قاعدة اختبار مرة واحدة (نفس خطوات db:create مع تغيير الاسم إلى lughati_test في TEST_DATABASE_URL).
2. من مجلد `backend`: `npm test` — تشمل مصادقة، صلاحيات (طالبة × بيانات طالبة أخرى = منع)، تراكمية التقييم، قيود الحضور، نجم الأسبوع، الحسابات، سجل التدقيق.

## الإنتاج
- `cd frontend && npm run build` ثم `cd ../backend && npm run build && npm start`
  — الخادم يقدّم الواجهة المبنية من نفس المنفذ.
- استخدمي https (عبر nginx مثلاً) ليصبح كوكي الجلسة `Secure`.
- نسخ احتياطي يومي: `pg_dump lughati > backup.sql` (جدوله عبر Task Scheduler).
- متغيرات `.env` تبقى على الخادم فقط، ولا تُرفع لأي مستودع.

-- مخطط منصة لغتي الكامل: جداول، علاقات، قيود، فهارس
CREATE TABLE students (
  id          serial PRIMARY KEY,
  code        varchar(30)  NOT NULL UNIQUE,          -- الرمز الثابت الذي يُربط به حساب الطالبة ولا يتغير
  name        varchar(120) NOT NULL,
  grade       varchar(20)  NOT NULL,
  section     varchar(10)  NOT NULL,                 -- الشعبة (class محجوزة في SQL)
  notes       text,
  active      boolean      NOT NULL DEFAULT true,
  created_at  timestamptz  NOT NULL DEFAULT now()
);

CREATE TABLE users (
  id            serial PRIMARY KEY,
  username      varchar(60)  NOT NULL UNIQUE,
  password_hash varchar(200) NOT NULL,                 -- bcrypt فقط، ولا تُعاد للواجهة أبداً
  role          varchar(10)  NOT NULL CHECK (role IN ('teacher','student','admin')),
  student_id    integer      UNIQUE REFERENCES students(id) ON DELETE CASCADE,
  active        boolean      NOT NULL DEFAULT true,
  token_version integer      NOT NULL DEFAULT 0,       -- يُرفع عند تغيير كلمة المرور فتُبطَل كل الجلسات القديمة
  created_at    timestamptz  NOT NULL DEFAULT now(),
  CONSTRAINT student_role_needs_student CHECK ((role = 'student') = (student_id IS NOT NULL))
);

-- التقييم تراكمي: لا يوجد UNIQUE على (student_id, eval_date) عمداً
CREATE TABLE evaluations (
  id            bigserial PRIMARY KEY,
  student_id    integer   NOT NULL REFERENCES students(id) ON DELETE CASCADE,
  eval_date     date      NOT NULL,
  attendance    varchar(8) NOT NULL CHECK (attendance IN ('present','excused','absent')),
  behavior      smallint CHECK (behavior BETWEEN 1 AND 5),
  participation smallint CHECK (participation BETWEEN 1 AND 5),
  homework      smallint CHECK (homework BETWEEN 1 AND 5),
  notes         text,
  demo          boolean   NOT NULL DEFAULT false,
  created_by    integer   REFERENCES users(id) ON DELETE SET NULL,
  created_at    timestamptz NOT NULL DEFAULT now(),
  CONSTRAINT absent_has_no_scores CHECK (
    attendance = 'present' OR (behavior IS NULL AND participation IS NULL AND homework IS NULL)
  )
);

CREATE TABLE worksheets (
  id          serial PRIMARY KEY,
  title       varchar(160) NOT NULL,
  description text,
  grade       varchar(20) NOT NULL,
  section     varchar(10) NOT NULL,
  work_date   date NOT NULL DEFAULT CURRENT_DATE,
  demo        boolean NOT NULL DEFAULT false,
  created_by  integer REFERENCES users(id) ON DELETE SET NULL,
  created_at  timestamptz NOT NULL DEFAULT now()
);
-- لا صفوف targets = الورقة لجميع طالبات الصف؛ وجود صف = معيّنة لطالبة بعينها
CREATE TABLE worksheet_targets (
  worksheet_id integer NOT NULL REFERENCES worksheets(id) ON DELETE CASCADE,
  student_id   integer NOT NULL REFERENCES students(id)  ON DELETE CASCADE,
  done         boolean NOT NULL DEFAULT false,
  updated_at   timestamptz NOT NULL DEFAULT now(),
  PRIMARY KEY (worksheet_id, student_id)
);

CREATE TABLE tests (
  id         serial PRIMARY KEY,
  title      varchar(160) NOT NULL,
  kind       varchar(8) NOT NULL CHECK (kind IN ('test','nafis')),  -- نافس ضمن نفس الجدول
  max_score  numeric(5,2) NOT NULL DEFAULT 10 CHECK (max_score > 0),
  test_date  date NOT NULL DEFAULT CURRENT_DATE,
  grade      varchar(20) NOT NULL,
  section    varchar(10) NOT NULL,
  demo       boolean NOT NULL DEFAULT false,
  created_by integer REFERENCES users(id) ON DELETE SET NULL,
  created_at timestamptz NOT NULL DEFAULT now()
);
CREATE TABLE results (
  id         bigserial PRIMARY KEY,
  test_id    integer NOT NULL REFERENCES tests(id)    ON DELETE CASCADE,
  student_id integer NOT NULL REFERENCES students(id) ON DELETE CASCADE,
  score      numeric(5,2) NOT NULL CHECK (score >= 0),
  note       text,
  updated_at timestamptz NOT NULL DEFAULT now(),
  UNIQUE (test_id, student_id)
);

CREATE TABLE plans (
  id         serial PRIMARY KEY,
  student_id integer NOT NULL REFERENCES students(id) ON DELETE CASCADE,
  title      varchar(160) NOT NULL,
  goal       text,
  actions    text,
  from_date  date,
  to_date    date,
  status     varchar(8) NOT NULL DEFAULT 'active' CHECK (status IN ('active','done')),
  demo       boolean NOT NULL DEFAULT false,
  created_by integer REFERENCES users(id) ON DELETE SET NULL,
  created_at timestamptz NOT NULL DEFAULT now()
);
CREATE TABLE plan_followups (
  id          bigserial PRIMARY KEY,
  plan_id     integer NOT NULL REFERENCES plans(id) ON DELETE CASCADE,
  follow_date date NOT NULL DEFAULT CURRENT_DATE,
  note        text NOT NULL,
  created_at  timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE skills (
  id          serial PRIMARY KEY,
  name        varchar(120) NOT NULL UNIQUE,
  description text,
  demo        boolean NOT NULL DEFAULT false,
  created_at  timestamptz NOT NULL DEFAULT now()
);
CREATE TABLE skill_levels (
  id         bigserial PRIMARY KEY,
  skill_id   integer NOT NULL REFERENCES skills(id)   ON DELETE CASCADE,
  student_id integer NOT NULL REFERENCES students(id) ON DELETE CASCADE,
  level      smallint NOT NULL CHECK (level BETWEEN 1 AND 5),
  note       text,
  updated_at timestamptz NOT NULL DEFAULT now(),
  UNIQUE (skill_id, student_id)
);

CREATE TABLE messages (
  id           bigserial PRIMARY KEY,
  student_id   integer NOT NULL REFERENCES students(id) ON DELETE CASCADE,
  sender       varchar(8) NOT NULL CHECK (sender IN ('teacher','student')),
  body         text NOT NULL,
  sent_at      timestamptz NOT NULL DEFAULT now(),
  seen_teacher boolean NOT NULL DEFAULT false,
  seen_student boolean NOT NULL DEFAULT false
);

CREATE TABLE weekly_star (                 -- «أميرتي هذا الأسبوع» (بداية الأسبوع = الأحد)
  id          serial PRIMARY KEY,
  week_start  date NOT NULL UNIQUE,
  week_label  varchar(40) NOT NULL,
  student_id  integer NOT NULL REFERENCES students(id) ON DELETE CASCADE,
  reason      text,
  demo        boolean NOT NULL DEFAULT false,
  created_by  integer REFERENCES users(id) ON DELETE SET NULL,
  created_at  timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE achievements (
  id          serial PRIMARY KEY,
  title       varchar(160) NOT NULL,
  description text,
  category    varchar(40) NOT NULL DEFAULT 'أخرى',
  ach_date    date NOT NULL DEFAULT CURRENT_DATE,
  visible     boolean NOT NULL DEFAULT true,   -- الظهور لبوابة الطالبات
  demo        boolean NOT NULL DEFAULT false,
  created_by  integer REFERENCES users(id) ON DELETE SET NULL,
  created_at  timestamptz NOT NULL DEFAULT now()
);

-- كل ملف مرفوع يمر من هنا: التنزيل يتم عبر /api/files/:id فقط، مع فحص ملكية في الخادم
CREATE TABLE files (
  id             bigserial PRIMARY KEY,
  stored_name    varchar(120) NOT NULL UNIQUE,   -- اسم عشوائي على القرص (لا يمكن تخمين مسار)
  original_name  varchar(255) NOT NULL,
  mime           varchar(120) NOT NULL,
  size_bytes     integer NOT NULL,
  kind           varchar(20) NOT NULL,           -- worksheet | achievement | logo
  student_id     integer REFERENCES students(id)     ON DELETE CASCADE,  -- خاص بطالبة محددة
  worksheet_id   integer REFERENCES worksheets(id)   ON DELETE CASCADE,
  achievement_id integer REFERENCES achievements(id) ON DELETE CASCADE,
  uploaded_by    integer REFERENCES users(id) ON DELETE SET NULL,
  created_at     timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE settings (
  key        varchar(60) PRIMARY KEY,
  value      jsonb NOT NULL,
  updated_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE audit_logs (
  id        bigserial PRIMARY KEY,
  at        timestamptz NOT NULL DEFAULT now(),
  user_id   integer,
  username  varchar(60),
  role      varchar(10),
  action    varchar(60) NOT NULL,   -- login / login_fail / evaluation.create / file.download ...
  entity    varchar(40),
  entity_id varchar(40),
  details   jsonb,
  ip        varchar(60)
);

CREATE INDEX idx_eval_student      ON evaluations(student_id, eval_date);
CREATE INDEX idx_eval_date         ON evaluations(eval_date);
CREATE INDEX idx_targets_student   ON worksheet_targets(student_id);
CREATE INDEX idx_results_student   ON results(student_id);
CREATE INDEX idx_plans_student     ON plans(student_id);
CREATE INDEX idx_levels_student    ON skill_levels(student_id);
CREATE INDEX idx_messages_student  ON messages(student_id, sent_at);
CREATE INDEX idx_files_student     ON files(student_id);
CREATE INDEX idx_audit_at          ON audit_logs(at DESC);

export const app = createApp();

let ready: Promise<void> | null = null;
async function setup() {
  await runMigrations();
  await q(`TRUNCATE audit_logs, files, achievements, weekly_star, messages, skill_levels, skills,
           plan_followups, plans, results, tests, worksheet_targets, worksheets, evaluations,
           users, students, settings RESTART IDENTITY CASCADE`);
  const h1 = bcrypt.hashSync('teacher-pass', 8);
  const h2 = bcrypt.hashSync('student-pass', 8);
  await q(`INSERT INTO students(id, code, name, grade, section) VALUES
           (1,'ST-001','طالبة اختبار أ','الرابع','أ'), (2,'ST-002','طالبة اختبار ب','الرابع','أ')`);
  await q(`INSERT INTO users(id, username, password_hash, role, student_id) VALUES
           (1,'teacher1',$1,'teacher',NULL), (2,'studentA',$2,'student',1), (3,'studentB',$2,'student',2)`, [h1, h2]);
}
export function ensureReady() { ready ??= setup(); return ready; }

export async function login(username: string, password: string): Promise<string> {
  const res = await request(app).post('/api/auth/login').send({ username, password });
  if (res.status !== 200) throw new Error('فشل تسجيل الدخول: ' + res.status + ' ' + JSON.stringify(res.body));
  return [].concat(res.headers['set-cookie'] as any).join('; ');
}

export const asStudentA = async () => login('studentA', 'student-pass');

  it('مسار محمي بدون جلسة ← 401', async () => {

  it('GET /api/auth/me بجلسة سليمة يعرف صاحبة الجلسة', async () => {
