//<<<FILE: backend/src/middleware/errors.ts>>>
import { Request, Response, NextFunction } from 'express';

export class HttpError extends Error {
  constructor(public status: number, message: string) { super(message); }
}
type H = (req: Request, res: Response, next: NextFunction) => unknown;
export const wrap = (fn: H) => (req: Request, res: Response, next: NextFunction) =>
  Promise.resolve(fn(req, res, next)).catch(next);

export function notFound(_req: Request, res: Response) { res.status(404).json({ error: 'المسار غير موجود' }); }
export function errorHandler(err: any, _req: Request, res: Response, _next: NextFunction) {
  const status = err instanceof HttpError ? err.status : 500;
  if (status >= 500) console.error(err);
  res.status(status).json({ error: err.message || 'حدث خطأ في الخادم' });
}

//<<<FILE: backend/src/middleware/auth.ts>>>
import { Request, Response, NextFunction } from 'express';
import jwt from 'jsonwebtoken';
import { env } from '../config';
import { q } from '../database/pool';
import { HttpError, wrap } from './errors';

export const COOKIE = 'lg_tok';
export interface AuthUser {
  id: number; username: string; role: 'teacher' | 'student' | 'admin';
  studentId: number | null; name?: string; tokenVersion: number;
}
declare global {
  namespace Express { interface Request { user?: AuthUser } }
}

export function setAuthCookie(res: Response, token: string) {
  res.cookie(COOKIE, token, {
    httpOnly: true, sameSite: 'lax', secure: env.nodeEnv === 'production',
    maxAge: 8 * 3600 * 1000, path: '/'
  });
}
export function clearAuthCookie(res: Response) { res.clearCookie(COOKIE, { path: '/' }); }

export const requireAuth = wrap(async (req: Request, _res: Response, next: NextFunction) => {
  const tok = (req as any).cookies?.[COOKIE];
  if (!tok) throw new HttpError(401, 'الجلسة منتهية، يرجى تسجيل الدخول');
  let p: any;
  try { p = jwt.verify(tok, env.jwtSecret); } catch { throw new HttpError(401, 'الجلسة منتهية، يرجى تسجيل الدخول'); }
  // الدور وحالة الحساب مصدرهما قاعدة البيانات في كل طلب — لا يُوثق بما يقوله المتصفح
  const { rows } = await q<any>(
    'SELECT u.id, u.username, u.role, u.active, u.token_version, u.student_id, s.name AS student_name ' +
    'FROM users u LEFT JOIN students s ON s.id = u.student_id WHERE u.id = $1', [p.uid]);
  const u = rows[0];
  if (!u || !u.active || u.token_version !== p.tv) throw new HttpError(401, 'الجلسة غير صالحة');
  req.user = {
    id: u.id, username: u.username, role: u.role, tokenVersion: u.token_version,
    studentId: u.student_id, name: u.student_name ?? undefined
  };
  next();
});

export const requireRole = (...roles: string[]) =>
  wrap((req: Request, _res: Response, next: NextFunction) => {
    if (!req.user) throw new HttpError(401, 'غير مصادق');
    if (!roles.includes(req.user.role)) throw new HttpError(403, 'غير مصرح بهذه العملية لدورك');
    next();
  });

/** تُمنع الطالبة من أي student_id ليس معرّفها — تُستدعى داخل الخدمات نفسها كخط دفاع ثانٍ */
export function assertStudentAccess(user: AuthUser, studentId: number) {
  if (user.role === 'student' && user.studentId !== studentId)
    throw new HttpError(403, 'غير مصرح بالوصول لهذه البيانات');
}

// ينشئ قاعدة البيانات إن لم توجد. لقاعدة الاختبار: npm run db:create -- test
import { Client } from 'pg';
import { env } from '../config';

async function main() {
  let target = env.databaseUrl;
  if (process.argv[2] === 'test') {
    if (!process.env.TEST_DATABASE_URL)
      throw new Error('قاعدة الاختبار تحتاج TEST_DATABASE_URL في ملف .env');
    target = process.env.TEST_DATABASE_URL;
  }
  const url = new URL(target);
  const dbName = decodeURIComponent(url.pathname.replace(/^\//, ''));
  if (!dbName) throw new Error('سلسلة الاتصال لا تحتوي اسم قاعدة البيانات');
  url.pathname = '/postgres'; // الاتصال بقاعدة النظام لإنشاء القاعدة الجديدة
  const c = new Client({ connectionString: url.toString() });
  await c.connect();
  const r = await c.query('SELECT 1 FROM pg_database WHERE datname = $1', [dbName]);
  if (!r.rowCount) {
    await c.query('CREATE DATABASE "' + dbName.replace(/"/g, '') + '"');
    console.log('✔ أُنشئت القاعدة:', dbName);
  } else {
    console.log('القاعدة موجودة مسبقاً:', dbName);
  }
  await c.end();
}

main().catch(e => { console.error('✖ فشل إنشاء قاعدة البيانات:', e.message); process.exit(1); });

//<<<FILE: backend/src/scripts/migrate.ts>>>
import fs from 'fs';
import path from 'path';
import { env } from '../config';
import { pool } from '../database/pool';

export async function runMigrations(dir = env.migrationsDir) {
  await pool.query('CREATE TABLE IF NOT EXISTS schema_migrations(name varchar(120) PRIMARY KEY, applied_at timestamptz NOT NULL DEFAULT now())');
  const files = fs.readdirSync(dir).filter(f => f.endsWith('.sql')).sort();
  for (const f of files) {
    const done = await pool.query('SELECT 1 FROM schema_migrations WHERE name = $1', [f]);
    if (done.rowCount) continue;
    await pool.query(fs.readFileSync(path.join(dir, f), 'utf8'));
    await pool.query('INSERT INTO schema_migrations(name) VALUES ($1)', [f]);
    console.log('✔ migration:', f);
  }
}

if (require.main === module) {
  runMigrations()
    .then(() => pool.end())
    .then(() => process.exit(0))
    .catch(e => { console.error('✖ فشل الترحيل:', e.message); process.exit(1); });
}

import fs from 'fs';
import path from 'path';
import { env } from '../config';
import { pool, q } from '../database/pool';
import { runMigrations } from './migrate';
import { hashPassword } from '../services/auth';

async function createAdmin() {
  const username = (process.env.ADMIN_USERNAME || 'admin').trim().toLowerCase();
  const password = process.env.ADMIN_PASSWORD || 'lughati1234';
  const { rowCount } = await q('SELECT 1 FROM users WHERE username = $1', [username]);
  if (rowCount) { console.log('حساب المعلمة موجود مسبقاً:', username); return; }
  await q("INSERT INTO users(username, password_hash, role) VALUES ($1,$2,'teacher')",
    [username, await hashPassword(password)]);
  console.log('✔ أُنشئ حساب المعلمة — اسم المستخدم: ' + username + ' (غيّري كلمة المرور فوراً من الإعدادات)');
}

async function main() {
  await runMigrations();
  const seedFile = path.join(env.seedsDir, 'seed.sql');
  if (fs.existsSync(seedFile)) {
    await pool.query(fs.readFileSync(seedFile, 'utf8')); // idempotent: ON CONFLICT DO NOTHING
    console.log('✔ seed.sql نُفّذ');
  } else {
    console.log('-- لا يوجد seed.sql — تم تجاوز البذر');
  }
  await createAdmin();
  await pool.end();
}

main().catch(e => { console.error('✖ فشل البذر:', e.message); process.exit(1); });

//<<<FILE: backend/src/scripts/import-students.ts>>>
// الاستخدام: npm run import:students -- ../database/seeds/students.csv
// كل سطر: الاسم,الصف,الشعبة (الرأس والأسطر التي تبدأ بـ # تُتجاهل، المكرر يُتخطى تلقائياً)
import fs from 'fs';
import path from 'path';
import { pool, q } from '../database/pool';
import { runMigrations } from './migrate';

async function main() {
  const file = process.argv[2];
  if (!file) throw new Error('الاستخدام: npm run import:students -- ملف.csv');
  await runMigrations();

  const lines = fs.readFileSync(path.resolve(file), 'utf8').split(/\r?\n/).map(l => l.trim()).filter(Boolean);
  let added = 0, skipped = 0;
  for (const [i, line] of lines.entries()) {
    if (line.startsWith('#')) continue;
    const cols = line.split(/[,;\t،]/).map(c => c.trim()).filter(Boolean);
    if (!cols.length) continue;
    const [name, gradeIn, sectionIn] = cols;
    if (i === 0 && /^(الاسم|name)$/i.test(name)) continue;
    const grade = gradeIn || 'الرابع';
    const section = sectionIn || 'أ';
    const ex = await q('SELECT 1 FROM students WHERE name = $1 AND grade = $2 AND section = $3', [name, grade, section]);
    if (ex.rowCount) { skipped++; continue; }
    const r = await q(
      "INSERT INTO students(code, name, grade, section) VALUES ('TMP-' || md5(random()::text), $1,$2,$3) RETURNING id",
      [name, grade, section]);
    await q("UPDATE students SET code = 'ST-' || lpad(id::text, 3, '0') WHERE id = $1", [r.rows[0].id]);
    added++;
  }
  console.log('أُضيف ' + added + ' طالبة، وتخطي ' + skipped + ' (مكرر/غير مكتمل)');
  await pool.end();
}

main().catch(e => { console.error('✖ فشل الاستيراد:', e.message); process.exit(1); });
