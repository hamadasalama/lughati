import type { ReactNode } from 'react';
import { Navigate, Route, Routes } from 'react-router-dom';
import { AuthProvider, useAuth } from './auth/AuthContext';
import Layout from './components/Layout';
import { ToastProvider } from './components/ui';
import Login from './pages/Login';
import Students from './pages/teacher/Students';
import StudentRecord from './pages/teacher/StudentRecord';
import Settings from './pages/teacher/Settings';
import Home from './pages/student/Home';
import MyEvaluations from './pages/student/MyEvaluations';
import MyMessages from './pages/student/MyMessages';
import MyAccount from './pages/student/MyAccount';

function Loading() {
  return <div style={{ padding: 40, textAlign: 'center', color: 'var(--mut)' }}>جارٍ التحميل…</div>;
}

function RootRedirect() {
  const { me, loading } = useAuth();
  if (loading) return <Loading />;
  if (!me) return <Navigate to="/login" replace />;
  return <Navigate to={me.role === 'student' ? '/s_home' : '/students'} replace />;
}

function Protected({ teacher, children }: { teacher?: boolean; children: ReactNode }) {
  const { me, loading } = useAuth();
  if (loading) return <Loading />;
  if (!me) return <Navigate to="/login" replace />;
  if (teacher && me.role === 'student') return <Navigate to="/s_home" replace />;
  if (!teacher && me.role !== 'student') return <Navigate to="/students" replace />;
  return <Layout>{children}</Layout>;
}

export default function App() {
  return (
    <ToastProvider>
      <AuthProvider>
        <Routes>
          <Route path="/login" element={<Login />} />
          <Route path="/" element={<RootRedirect />} />
          <Route path="/students" element={<Protected teacher><Students /></Protected>} />
          <Route path="/students/:id" element={<Protected teacher><StudentRecord /></Protected>} />
          <Route path="/settings" element={<Protected teacher><Settings /></Protected>} />
          <Route path="/s_home" element={<Protected><Home /></Protected>} />
          <Route path="/s_evals" element={<Protected><MyEvaluations /></Protected>} />
          <Route path="/s_msgs" element={<Protected><MyMessages /></Protected>} />
          <Route path="/s_account" element={<Protected><MyAccount /></Protected>} />
          <Route path="*" element={<RootRedirect />} />
        </Routes>
      </AuthProvider>
    </ToastProvider>
  );
}

export class ApiError extends Error {
  status: number;
  constructor(status: number, message: string) {
    super(message);
    this.status = status;
  }
}

export async function api(path: string, opts: RequestInit = {}): Promise<any> {
  const init: RequestInit = { credentials: 'include', ...opts };
  if (opts.body && !(opts.body instanceof FormData)) {
    init.headers = { 'Content-Type': 'application/json', ...((opts.headers as any) || {}) };
  }
  let res: Response;
  try {
    res = await fetch('/api' + path, init);
  } catch {
    throw new ApiError(0, 'تعذر الاتصال بالخادم — تأكدي من تشغيله (npm run dev:server)');
  }
  if (res.status === 204) return undefined;
  let data: any = null;
  try { data = await res.json(); } catch { data = null; }
  if (!res.ok) throw new ApiError(res.status, (data && data.error) || ('خطأ ' + res.status));
  return data;
}

export function qs(params: Record<string, string | number | undefined | null>): string {
  const parts: string[] = [];
  for (const [k, v] of Object.entries(params)) {
    if (v !== undefined && v !== null && v !== '') {
      parts.push(encodeURIComponent(k) + '=' + encodeURIComponent(String(v)));
    }
  }
  return parts.length ? '?' + parts.join('&') : '';
}

import { createContext, useCallback, useContext, useRef, useState, type ReactNode } from 'react';

interface Toast { id: number; text: string; err?: boolean; }
interface ToastCtx { push: (text: string, err?: boolean) => void; }

const TCtx = createContext<ToastCtx>({ push: () => {} });
let seq = 1;

export function ToastProvider({ children }: { children: ReactNode }) {
  const [items, setItems] = useState<Toast[]>([]);
  const push = useCallback((text: string, err?: boolean) => {
    const id = seq++;
    setItems((xs) => [...xs, { id, text, err }]);
    setTimeout(() => setItems((xs) => xs.filter((t) => t.id !== id)), 3400);
  }, []);
  return (
    <TCtx.Provider value={{ push }}>
      {children}
      <div className="toasts">
        {items.map((t) => (
          <div key={t.id} className={'toast' + (t.err ? ' err' : '')}>{t.text}</div>
        ))}
      </div>
    </TCtx.Provider>
  );
}

export function useToast() {
  return useContext(TCtx);
}

export function Modal({ title, onClose, wide, children }: {

  children: ReactNode;
}) {
  const ref = useRef<HTMLDivElement>(null);

      }}
    >
      <div className={'modal' + (wide ? ' wide' : '')} ref={ref}>
        <div className="mhead">
          <h3>{title}</h3>

        </div>

    if (!sid) return;
    try { setMsgs(await api('/messages/' + sid)); } catch { /* تجاهل */ }
  }, [sid]);

    load();
    const t = setInterval(load, 8000);
    return () => clearInterval(t);
  }, [load]);

    const el = boxRef.current;
    if (el) el.scrollTop = el.scrollHeight;
  }, [msgs]);

    if (!body) return;
    try {
      await api('/messages', { method: 'POST', body: JSON.stringify({ student_id: sid, body }) });
      setText('');
      load();
    } catch (err) {
      push(err instanceof ApiError ? err.message : 'خطأ غير متوقع', true);
    }
  }

  return (
    <div>
      <div className="page-head"><h2>رسائلي</h2></div>
      <div className="card">
        <div className="chat">
          <div className="chatlist" ref={boxRef}>
            {msgs.length === 0 && (
              <div className="empty">لا رسائل بعد — اكتبي رسالتك الأولى لمعلمة المادة.</div>
            )}
            {msgs.map((m, i) => (
              <div key={m.id ?? i} className={'mb ' + (m.sender === 'student' ? 's' : 't')}>
                {m.body}
                <div className="meta">
                  {new Date(m.sent_at).toLocaleString('ar', { dateStyle: 'short', timeStyle: 'short' } as Intl.DateTimeFormatOptions)}
                </div>
              </div>
            ))}
          </div>
          <form className="chatbar" onSubmit={send}>
            <input
              value={text}
              onChange={(e) => setText(e.target.value)}
              placeholder="اكتبي رسالتك…"
              maxLength={4000}
            />
            <button className="btn">إرسال</button>
          </form>
        </div>
      </div>
    </div>
  );
}

import { useState, type FormEvent } from 'react';
import { useNavigate } from 'react-router-dom';
import { api, ApiError } from '../../api/client';
import { useAuth } from '../../auth/AuthContext';
import { useToast } from '../../components/ui';

export default function MyAccount() {
  const { me } = useAuth();
  const [form, setForm] = useState({ current: '', nw: '', confirm: '' });
  const { push } = useToast();
  const nav = useNavigate();

  async function submit(e: FormEvent) {
    e.preventDefault();
    if (form.nw !== form.confirm) { push('كلمتا المرور الجديدتان غير متطابقتين', true); return; }
    if (form.nw.length < 6) { push('كلمة المرور الجديدة يجب ألا تقل عن 6 أحرف', true); return; }
    try {
      await api('/auth/change-password', {
        method: 'POST',
        body: JSON.stringify({ current: form.current, new: form.nw }),
      });
      push('تم تغيير كلمة مرورك بنجاح — يُرجى تسجيل الدخول من جديد');
      setTimeout(async () => {
        try { await api('/auth/logout', { method: 'POST' }); } catch { /* تجاهل */ }
        nav('/login');
      }, 1200);
    } catch (err) {
      push(err instanceof ApiError ? err.message : 'خطأ غير متوقع', true);
    }
  }

  return (
    <div>
      <div className="page-head"><h2>حسابي</h2></div>
      <div className="card" style={{ maxWidth: 560 }}>
        <p>
          اسم المستخدم: <b>{me?.username}</b>
          {me?.name ? <> — {me.name}</> : null}
        </p>
        <p className="muted">
          كلمة مرورك مرتبطة بحسابك الشخصي فقط، ولا يمكن لأحد غيرك رؤيتها أو تغييرها.
          تغييرها من هنا يُبطل أي جلسة مفتوحة بحسابك.
        </p>
        <form onSubmit={submit}>
          <div className="fld">
            <label>كلمة المرور الحالية</label>
            <input type="password" value={form.current} required onChange={(e) => setForm({ ...form, current: e.target.value })} />
          </div>
          <div className="fld">
            <label>كلمة المرور الجديدة (6 أحرف على الأقل)</label>
            <input type="password" value={form.nw} required onChange={(e) => setForm({ ...form, nw: e.target.value })} />
          </div>
          <div className="fld">
            <label>تأكيد كلمة المرور الجديدة</label>
            <input type="password" value={form.confirm} required onChange={(e) => setForm({ ...form, confirm: e.target.value })} />
          </div>
          <button className="btn">حفظ كلمة المرور الجديدة</button>
        </form>
      </div>
    </div>
  );
}
