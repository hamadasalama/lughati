import type { ReactNode } from 'react';
import { Navigate, Route, Routes } from 'react-router-dom';
import { AuthProvider, useAuth } from './auth/AuthContext';
import Layout from './components/Layout';
import { ToastProvider } from './components/ui';
import Login from './pages/Login';
import Students from './pages/teacher/Students';
import StudentRecord from './pages/teacher/StudentRecord';
import Worksheets from './pages/teacher/Worksheets';
import Tests from './pages/teacher/Tests';
import Plans from './pages/teacher/Plans';
import Skills from './pages/teacher/Skills';
import Msgs from './pages/teacher/Msgs';
import Ach from './pages/teacher/Ach';
import Alerts from './pages/teacher/Alerts';
import Analytics from './pages/teacher/Analytics';
import Compare from './pages/teacher/Compare';
import Print from './pages/teacher/Print';
import Settings from './pages/teacher/Settings';
import Home from './pages/student/Home';
import MyEvaluations from './pages/student/MyEvaluations';
import MyResults from './pages/student/MyResults';
import MyWorksheets from './pages/student/MyWorksheets';
import MyPlan from './pages/student/MyPlan';
import MySkills from './pages/student/MySkills';
import MyAchievements from './pages/student/MyAchievements';
import MyMessages from './pages/student/MyMessages';
import MyAccount from './pages/student/MyAccount';

function Loading() {
  return <div style={{ padding: 40, textAlign: 'center', color: 'var(--mut)' }}>جارٍ التحميل…</div>;
}

function RootRedirect() {
  const { me, loading } = useAuth();
  if (loading) return <Loading />;
  if (!me) return <Navigate to="/login" replace />;
  return <Navigate to={me.role === 'student' ? '/s_home' : '/students'} replace />;
}

function Protected({ teacher, children }: { teacher?: boolean; children: ReactNode }) {
  const { me, loading } = useAuth();
  if (loading) return <Loading />;
  if (!me) return <Navigate to="/login" replace />;
  if (teacher && me.role === 'student') return <Navigate to="/s_home" replace />;
  if (!teacher && me.role !== 'student') return <Navigate to="/students" replace />;
  return <Layout>{children}</Layout>;
}

export default function App() {
  return (
    <ToastProvider>
      <AuthProvider>
        <Routes>
          <Route path="/login" element={<Login />} />
          <Route path="/" element={<RootRedirect />} />

          <Route path="/students" element={<Protected teacher><Students /></Protected>} />
          <Route path="/students/:id" element={<Protected teacher><StudentRecord /></Protected>} />
          <Route path="/worksheets" element={<Protected teacher><Worksheets /></Protected>} />
          <Route path="/tests" element={<Protected teacher><Tests /></Protected>} />
          <Route path="/plans" element={<Protected teacher><Plans /></Protected>} />
          <Route path="/skills" element={<Protected teacher><Skills /></Protected>} />
          <Route path="/msgs" element={<Protected teacher><Msgs /></Protected>} />
          <Route path="/ach" element={<Protected teacher><Ach /></Protected>} />
          <Route path="/alerts" element={<Protected teacher><Alerts /></Protected>} />
          <Route path="/analytics" element={<Protected teacher><Analytics /></Protected>} />
          <Route path="/compare" element={<Protected teacher><Compare /></Protected>} />
          <Route path="/print" element={<Protected teacher><Print /></Protected>} />
          <Route path="/settings" element={<Protected teacher><Settings /></Protected>} />

          <Route path="/s_home" element={<Protected><Home /></Protected>} />
          <Route path="/s_evals" element={<Protected><MyEvaluations /></Protected>} />
          <Route path="/s_results" element={<Protected><MyResults /></Protected>} />
          <Route path="/s_work" element={<Protected><MyWorksheets /></Protected>} />
          <Route path="/s_plan" element={<Protected><MyPlan /></Protected>} />
          <Route path="/s_skills" element={<Protected><MySkills /></Protected>} />
          <Route path="/s_ach" element={<Protected><MyAchievements /></Protected>} />
          <Route path="/s_msgs" element={<Protected><MyMessages /></Protected>} />
          <Route path="/s_account" element={<Protected><MyAccount /></Protected>} />

          <Route path="*" element={<RootRedirect />} />
        </Routes>
      </AuthProvider>
    </ToastProvider>
  );
}

import { useCallback, useEffect, useState, type FormEvent } from 'react';
import { api, ApiError, qs } from '../../api/client';
import type { Student } from '../../api/types';
import Icon from '../../components/Icon';
import { Modal, useToast } from '../../components/ui';

const EMPTY = { title: '', description: '', grade: 'الرابع', section: 'أ', due_date: '', ids: [] as number[] };

export default function Worksheets() {
  const [rows, setRows] = useState<any[]>([]);
  const [grades, setGrades] = useState<string[]>(['الرابع', 'الخامس', 'السادس']);
  const [sections, setSections] = useState<string[]>(['أ', 'ب']);
  const [grade, setGrade] = useState('');
  const [section, setSection] = useState('');
  const [show, setShow] = useState(false);
  const [form, setForm] = useState<any>({ ...EMPTY });
  const [file, setFile] = useState<File | undefined>(undefined);
  const [track, setTrack] = useState<any | null>(null);
  const [tmap, setTmap] = useState<Record<number, boolean>>({});
  const [pool, setPool] = useState<Student[]>([]);
  const { push } = useToast();

  const load = useCallback(async () => {
    setRows(await api('/worksheets' + qs({ grade, section })));
  }, [grade, section]);

  useEffect(() => { load().catch((e: any) => push(e.message || 'خطأ', true)); }, [load, push]);

  useEffect(() => {
    api('/settings').then((c) => {
      if (Array.isArray(c.grades) && c.grades.length) setGrades(c.grades);
      if (Array.isArray(c.classes) && c.classes.length) setSections(c.classes);
    }).catch(() => {});
  }, []);

  useEffect(() => {
    if (!show) return;
    api('/students' + qs({ grade: form.grade, section: form.section })).then(setPool).catch(() => setPool([]));
  }, [show, form.grade, form.section]);

  useEffect(() => {
    if (!track) { setPool([]); setTmap({}); return; }
    api('/students' + qs({ grade: track.grade, section: track.section })).then(setPool).catch(() => setPool([]));
    const m: Record<number, boolean> = {};
    if (Array.isArray(track.targets)) {
      for (const t of track.targets) m[Number(t.student_id)] = !!t.done;
    }
    setTmap(m);
  }, [track]);

  async function create(e: FormEvent) {
    e.preventDefault();
    const fd = new FormData();
    fd.append('title', form.title);
    fd.append('description', form.description || '');
    fd.append('grade', form.grade);
    fd.append('section', form.section);
    if (form.due_date) fd.append('due_date', form.due_date);
    fd.append('ids', JSON.stringify(form.ids));
    if (file) fd.append('file', file);
    try {
      await api('/worksheets', { method: 'POST', body: fd });
      push(form.ids.length ? 'أُنشئت الورقة وأُسندت للطالبات المحددات' : 'أُنشئت الورقة لكل طالبات الصف والشعبة');
      setShow(false);
      setForm({ ...EMPTY });
      setFile(undefined);
      load();
    } catch (err) {
      push(err instanceof ApiError ? err.message : 'خطأ غير متوقع', true);
    }
  }

  async function remove(w: any) {
    if (!confirm('حذف «' + w.title + '» مع مرفقاتها وإسناداتها؟')) return;
    try {
      await api('/worksheets/' + w.id, { method: 'DELETE' });
      push('حُذفت الورقة');
      load();
    } catch (err) {
      push(err instanceof ApiError ? err.message : 'خطأ غير متوقع', true);
    }
  }

  async function mark(sid: number, done: boolean) {
    try {
      await api('/worksheets/' + track.id + '/status', {
        method: 'PATCH',
        body: JSON.stringify({ student_id: sid, done }),
      });
      setTmap({ ...tmap, [sid]: done });
      push(done ? 'عُلّمت كمنجَزة' : 'أُلغيت التعليم');
    } catch (err) {
      push(err instanceof ApiError ? err.message : 'خطأ غير متوقع', true);
    }
  }

  function tstate(sid: number): 'done' | 'not' | 'unknown' {
    if (tmap[sid] === undefined) return 'unknown';
    return tmap[sid] ? 'done' : 'not';
  }

  return (
    <div>
      <div className="page-head">
        <h2>أوراق العمل</h2>
        <button className="btn" onClick={() => { setForm({ ...EMPTY }); setFile(undefined); setShow(true); }}>
          <Icon name="plus" size={15} /> ورقة جديدة
        </button>
      </div>

      <div className="toolbar">
        <div className="fld">
          <label>الصف</label>
          <select value={grade} onChange={(e) => setGrade(e.target.value)}>
            <option value="">كل الصفوف</option>
            {grades.map((g) => <option key={g}>{g}</option>)}
          </select>
        </div>
        <div className="fld">
          <label>الشعبة</label>
          <select value={section} onChange={(e) => setSection(e.target.value)}>
            <option value="">كل الشعب</option>
            {sections.map((s) => <option key={s}>{s}</option>)}
          </select>
        </div>
      </div>

      <div className="card">
        {rows.length === 0 ? <div className="empty">لا أوراق عمل بعد.</div> : (
          <table>
            <thead>
              <tr><th>العنوان</th><th>الصف / الشعبة</th><th>آخر تسليم</th><th>المرفق</th><th>إجراءات</th></tr>
            </thead>
            <tbody>
              {rows.map((w) => (
                <tr key={w.id}>
                  <td>
                    <b>{w.title}</b>
                    {w.description ? <div className="muted" style={{ fontSize: 14 }}>{w.description}</div> : null}
                  </td>
                  <td>{w.grade} / {w.section}</td>
                  <td className="num">{w.due_date || '—'}</td>
                  <td>
                    {w.file_id
                      ? <a className="btn ghost sm" href={'/api/files/' + w.file_id} target="_blank" rel="noreferrer"><Icon name="download" size={14} /> تنزيل</a>
                      : '—'}
                  </td>
                  <td>
                    <div className="row-actions">
                      <button className="btn ghost sm" onClick={() => setTrack(w)}>رصد الإنجاز</button>
                      <button className="btn danger sm" onClick={() => remove(w)}>حذف</button>
                    </div>
                  </td>
                </tr>
              ))}
            </tbody>
          </table>
        )}
      </div>

      {show && (
        <Modal title="ورقة عمل جديدة" onClose={() => setShow(false)} wide>
          <form onSubmit={create}>
            <div className="form-grid">
              <div className="fld full">
                <label>عنوان الورقة</label>
                <input value={form.title} onChange={(e) => setForm({ ...form, title: e.target.value })} required />
              </div>
              <div className="fld full">
                <label>وصف / تعليمات</label>
                <textarea value={form.description || ''} onChange={(e) => setForm({ ...form, description: e.target.value })} />
              </div>
              <div className="fld">
                <label>الصف</label>
                <select value={form.grade} onChange={(e) => setForm({ ...form, grade: e.target.value, ids: [] })}>
                  {grades.map((g) => <option key={g}>{g}</option>)}
                </select>
              </div>
              <div className="fld">
                <label>الشعبة</label>
                <select value={form.section} onChange={(e) => setForm({ ...form, section: e.target.value, ids: [] })}>
                  {sections.map((s) => <option key={s}>{s}</option>)}
                </select>
              </div>
              <div className="fld">
                <label>آخر موعد للتسليم (اختياري)</label>
                <input type="date" value={form.due_date} onChange={(e) => setForm({ ...form, due_date: e.target.value })} />
              </div>
              <div className="fld">
                <label>مرفق (اختياري — حتى 6MB)</label>
                <input type="file" onChange={(e) => setFile(e.target.files?.[0])} />
              </div>
            </div>
            {pool.length > 0 && (
              <div className="fld">
                <label>إسناد لطالبات محددات (اتركيها فارغة لتشمل كل طالبات {form.grade}/{form.section})</label>
                <div style={{ maxHeight: 160, overflowY: 'auto', border: '1px solid var(--line)', borderRadius: 10, padding: '8px 10px', display: 'flex', flexWrap: 'wrap', gap: 6 }}>
                  {pool.map((s) => (
                    <label key={s.id} style={{ display: 'inline-flex', alignItems: 'center', gap: 5, fontSize: 15, background: form.ids.includes(s.id) ? 'var(--pk3)' : 'transparent', borderRadius: 8, padding: '2px 8px', cursor: 'pointer' }}>
                      <input
                        type="checkbox"
                        checked={form.ids.includes(s.id)}
                        onChange={(e) => setForm({
                          ...form,
                          ids: e.target.checked ? [...form.ids, s.id] : form.ids.filter((x: number) => x !== s.id),
                        })}
                      />
                      {s.name}
                    </label>
                  ))}
                </div>
              </div>
            )}
            <button className="btn">إنشاء الورقة</button>
          </form>
        </Modal>
      )}

      {track && (
        <Modal title={'رصد الإنجاز — ' + track.title} onClose={() => setTrack(null)} wide>
          {pool.length === 0 ? <div className="empty">لا طالبات في هذا الصف/الشعبة.</div> : (
            <table>
              <thead><tr><th>الطالبة</th><th>الحالة</th><th>تعليم</th></tr></thead>
              <tbody>
                {pool.map((s) => {
                  const st = tstate(s.id);
                  return (
                    <tr key={s.id}>
                      <td>{s.name}</td>
                      <td>
                        {st === 'done' ? <span className="badge ok">أُنجزت</span>
                          : st === 'not' ? <span className="badge">لم تُنجز</span>
                          : <span className="badge warn">غير مسجّل</span>}
                      </td>
                      <td>
                        <div className="row-actions">
                          <button className="btn ghost sm" disabled={st === 'done'} onClick={() => mark(s.id, true)}>أُنجزت</button>
                          <button className="btn ghost sm" disabled={st === 'not'} onClick={() => mark(s.id, false)}>إلغاء</button>
                        </div>
                      </td>
                    </tr>
                  );
                })}
              </tbody>
            </table>
          )}

function todayStr(): string {
  const d = new Date();
  const p = (n: number) => String(n).padStart(2, '0');
  return d.getFullYear() + '-' + p(d.getMonth() + 1) + '-' + p(d.getDate());
}

  const [sel, setSel] = useState<number | null>(null);
  const [name, setName] = useState('');
  const [vals, setVals] = useState<Record<number, number | null>>({});
  const [dirty, setDirty] = useState<Set<number>>(new Set());
  const { push } = useToast();

  async function load() {
    const [sk, stus] = await Promise.all([api('/skills'), api('/students')]);
    const list = Array.isArray(sk) ? sk : [];
    setRows(list);
    setStudents(stus);
    if (list.length && sel == null) setSel(list[0].id);
  }

  useEffect(() => { load().catch((e: any) => push(e.message || 'خطأ', true)); }, [push]);

  useEffect(() => {
    if (sel == null) { setVals({}); setDirty(new Set()); return; }
    const sk = rows.find((r) => r.id === sel);
    const v: Record<number, number | null> = {};
    if (sk && Array.isArray(sk.levels)) {
      for (const l of sk.levels) v[Number(l.student_id)] = Number(l.level);
    }
    setVals(v);
    setDirty(new Set());
  }, [sel, rows]);

  async function addSkill(e: FormEvent) {
    e.preventDefault();
    if (!name.trim()) return;
    try {
      await api('/skills', { method: 'POST', body: JSON.stringify({ name: name.trim() }) });
      push('أُضيفت المهارة');
      setName('');
      load();
    } catch (err) {
      push(err instanceof ApiError ? err.message : 'خطأ غير متوقع', true);
    }
  }

  function setLevel(sid: number, lv: number) {
    setVals({ ...vals, [sid]: lv });
    setDirty(new Set(dirty).add(sid));
  }

  async function saveLevels() {
    if (sel == null) return;
    const levels = [...dirty]
      .map((sid) => ({ student_id: sid, level: vals[sid] }))
      .filter((r) => r.level != null);
    if (!levels.length) { push('لا تغييرات لحفظها', true); return; }
    try {
      await api('/skills/' + sel + '/levels', { method: 'PUT', body: JSON.stringify({ levels, rows: levels }) });
      push('حُفظت المستويات');
      setDirty(new Set());
      load();
    } catch (err) {
      push(err instanceof ApiError ? err.message : 'خطأ غير متوقع', true);
    }
  }

  return (
    <div>
      <div className="page-head"><h2>المهارات بإتقان</h2></div>

          <select value={sel ?? ''} onChange={(e) => setSel(Number(e.target.value))}>

                        >{n}</button>
                      </td>
                    ))}
                  </tr>
                ))}
              </tbody>
            </table>
            <button className="btn" style={{ marginTop: 12 }} onClick={saveLevels} disabled={dirty.size === 0}>
              حفظ المستويات{dirty.size ? ' (' + dirty.size + ')' : ''}
            </button>
          </>
        )}
      </div>
    </div>
  );

function todayStr(): string {
  const d = new Date();
  const p = (n: number) => String(n).padStart(2, '0');
  return d.getFullYear() + '-' + p(d.getMonth() + 1) + '-' + p(d.getDate());
}

}

interface Item { name: string; cls: string; kind: string; text: string; }

export default function Alerts() {
  const [cfg, setCfg] = useState<any>({});
  const [items, setItems] = useState<Item[] | null>(null);
  const [noEvalNames, setNoEvalNames] = useState<string[]>([]);
  const [busy, setBusy] = useState(false);
  const { push } = useToast();

  useEffect(() => {
    api('/settings').then(setCfg).catch(() => {});
  }, []);

  const a = cfg?.alerts || {};
  const lowThreshold = Number(a.lowThreshold) || 2;
  const lowStreak = Number(a.lowStreak) || 3;
  const absenceDays = Number(a.absenceDays) || 3;
  const noEvalDays = Number(a.noEvalDays) || 14;

  async function analyze() {
    setBusy(true);
    setItems(null);
    try {
      const [stus, c] = await Promise.all([api('/students'), api('/settings')]);
      setCfg(c);
      const al = c?.alerts || {};
      const lowT = Number(al.lowThreshold) || 2;
      const lowS = Number(al.lowStreak) || 3;
      const absD = Number(al.absenceDays) || 3;
      const noD = Number(al.noEvalDays) || 14;
      const active: Student[] = stus.filter((s: Student) => s.active);
      const res = await Promise.all(active.map((s) =>
        api('/evaluations' + qs({ student_id: s.id }))
          .then((evs) => ({ s, evs: Array.isArray(evs) ? evs : [] }))
          .catch(() => ({ s, evs: [] as any[] }))
      ));
      const out: Item[] = [];
      const none: string[] = [];
      const today = Date.now();
      for (const { s, evs } of res) {
        const cls = s.grade + '/' + s.section;
        const sorted = [...evs].sort((x: any, y: any) => (x.eval_date < y.eval_date ? -1 : 1));
        const absent = sorted.filter((v: any) => v.attendance === 'absent').length;
        if (absent >= absD) {
          out.push({ name: s.name, cls, kind: 'غياب', text: 'غياب متكرر (' + absent + ' أيام) — يوصى بالتواصل مع الأسرة لدعم حضورها.' });
        }
        if (sorted.length === 0) {
          none.push(s.name);
        } else {
          const lastD = Date.parse(sorted[sorted.length - 1].eval_date + 'T00:00:00Z');
          const days = Math.floor((today - lastD) / 86400000);
          if (days >= noD) {
            out.push({ name: s.name, cls, kind: 'متابعة تقييم', text: 'لم تُسجَّل لها تقييمات منذ ' + days + ' يوماً — يوصى بمراجعة سجلها.' });
          }
        }
        let streak = 0;
        for (let i = sorted.length - 1; i >= 0; i--) {
          const v: any = sorted[i];
          if (v.attendance !== 'present') break;
          const av = ((Number(v.behavior) || 0) + (Number(v.participation) || 0) + (Number(v.homework) || 0)) / 3;
          if (av < lowT) streak++; else break;
        }
        if (streak >= lowS) {
          out.push({ name: s.name, cls, kind: 'مستوى', text: 'انخفاض في درجاتها في آخر ' + streak + ' تقييمات متتالية — تحتاج إلى متابعة وتعزيز.' });
        }
      }
      out.sort((x, y) => x.name.localeCompare(y.name, 'ar'));
      setItems(out);
      setNoEvalNames(none);
    } catch (err: any) {
      push(err?.message || 'خطأ غير متوقع', true);
    } finally {
      setBusy(false);
    }
  }

  return (
    <div>
      <div className="page-head">
        <h2>التنبيهات والتوصيات</h2>
        <button className="btn" onClick={analyze} disabled={busy}>{busy ? 'جارٍ التحليل…' : 'تحليل الوضع الحالي'}</button>
      </div>

      <div className="card">
        <h3>معايير التنبيه (من الإعدادات)</h3>
        <div className="muted">
          {lowStreak} تقييمات متتالية بمتوسط أقل من {lowThreshold} · غياب {absenceDays} أيام أو أكثر · بلا تقييم منذ {noEvalDays} يوماً أو أكثر
        </div>
        <p className="muted" style={{ fontSize: 14, marginBottom: 0 }}>
          التوصيات صياغات بنّاءة للاسترشاد وليست أحكاماً نهائية على الطالبات.
        </p>
      </div>

      {items === null ? (
        <div className="empty">انقري «تحليل الوضع الحالي» لعرض التوصيات.</div>
      ) : items.length === 0 ? (
        <div className="card"><div className="empty">لا توصيات حالياً وفق المعايير المحددة 🌿</div></div>
      ) : (
        <div className="card">

import { useEffect, useState } from 'react';
import { api } from '../../api/client';
import { useToast } from '../../components/ui';

function pick(row: any, keys: string[]): any {
  for (const k of keys) {
    if (row[k] !== undefined && row[k] !== null && row[k] !== '') return row[k];
  }
  return null;
}

function colorForScore(v: number): string {
  return v >= 4.5 ? '#1a5c42' : v >= 3.5 ? '#4c8a63' : v >= 2.5 ? '#c9a23f' : v >= 1.5 ? '#cf8a4a' : '#b8503c';
}

function rowAvg(r: any, keys: string[]): number | null {
  const v = pick(r, keys);
  return v == null || v === '' ? null : Number(v);
}

function avgOfRow(r: any): number | null {
  const xs = [
    rowAvg(r, ['behavior', 'behavior_avg']),
    rowAvg(r, ['participation', 'participation_avg']),
    rowAvg(r, ['homework', 'homework_avg']),
  ].filter((x): x is number => x != null && !Number.isNaN(x));
  return xs.length ? xs.reduce((a, b) => a + b, 0) / xs.length : null;
}

function trendBadge(r: any) {
  const t = r?.trend;
  if (!t || !t.enough) return <span className="muted">—</span>;
  const s = Number(t.slope);
  if (s > 0) return <span className="badge ok">↑ تتحسن</span>;
  if (s < 0) return <span className="badge warn">↓ تحتاج تعزيزاً</span>;
  return <span className="badge">→ مستقرة</span>;
}

function HeatCell({ v, plain }: { v: number | null; plain?: boolean }) {
  if (v == null || Number.isNaN(v)) return <td className="num" style={{ color: 'var(--mut)' }}>—</td>;
  if (plain) return <td className="num"><b>{v.toFixed(2)}</b></td>;
  const bg = colorForScore(v);
  return (
    <td
      className="num"
      style={{ background: bg, color: '#fff', printColorAdjust: 'exact', WebkitPrintColorAdjust: 'exact' } as any}
    >
      {v.toFixed(2)}
    </td>
  );
}

export default function Analytics() {
  const [rows, setRows] = useState<any[] | null>(null);
  const [err, setErr] = useState('');
  const [grade, setGrade] = useState('');
  const [section, setSection] = useState('');
  const [grades, setGrades] = useState<string[]>(['الرابع', 'الخامس', 'السادس']);
  const [sections, setSections] = useState<string[]>(['أ', 'ب']);
  const { push } = useToast();

  useEffect(() => {
    api('/reports/summary')
      .then((d: any) => setRows(Array.isArray(d) ? d : Array.isArray(d?.rows) ? d.rows : []))
      .catch((e: any) => setErr(e.message || 'خطأ'));
    api('/settings').then((c) => {
      if (Array.isArray(c.grades) && c.grades.length) setGrades(c.grades);
      if (Array.isArray(c.classes) && c.classes.length) setSections(c.classes);
    }).catch(() => {});
  }, []);

  if (err) return <div className="card"><div className="empty">{err}</div></div>;
  if (rows === null) return <div className="empty">جارٍ التحميل…</div>;

  const view = rows.filter((r) =>
    (!grade || String(pick(r, ['grade'])) === grade) &&
    (!section || String(pick(r, ['section'])) === section)
  );

  const tot = { present: 0, excused: 0, absent: 0 };
  for (const r of rows) {
    tot.present += Number(pick(r, ['present', 'present_count'])) || 0;
    tot.excused += Number(pick(r, ['excused', 'excused_count'])) || 0;
    tot.absent += Number(pick(r, ['absent', 'absent_count'])) || 0;

  const { me } = useAuth();

  const [kind, setKind] = useState<'all' | 'test' | 'nafis'>('all');

    api('/reports/student/' + sid).then(setRep).catch((e: any) => push(e.message || 'خطأ', true));

          <button className={kind === 'test' ? 'on' : ''} onClick={() => setKind('test')}>اختبارات صفية</button>

# منصة لغتي — تقييم طالبات مادة اللغة العربية

تطبيق ويب متكامل (Full-Stack): واجهة React + TypeScript، خادم Node.js/Express + TypeScript،
قاعدة بيانات PostgreSQL حقيقية، مصادقة بجلسات آمنة وصلاحيات مفروضة في الخادم.

**إعداد وتنفيذ: أ. مها الحمادي**
مديرة المدرسة: أ. بدور الجارد

---

## المتطلبات
- Node.js 18 أو أحدث
- PostgreSQL 14 أو أحدث يعمل محلياً

## التشغيل للتطوير (كل الأوامر من جذر المشروع)

1) **تثبيت تبعيات المشروعين معاً** (npm workspaces — مرة واحدة فقط من الجذر):

       npm install

2) **إنشاء ملف البيئة**: انسخي `.env.example` إلى `.env` وعدّلي:
   - `DATABASE_URL`: بيانات اتصالك بـ PostgreSQL.
     تنبيه: كلمة مرور تحتوي رموزاً مثل `@` أو `:` تفسد الرابط — استخدمي حروفاً وأرقاماً فقط.
   - `JWT_SECRET`: أي نص عشوائي طويل.

3) **إنشاء قاعدة البيانات والجداول والبيانات الأساسية:**

       npm run db:create
       npm run migrate
       npm run seed

   يُنشئ الـ seed حساب المعلمة الأول:
   - اسم المستخدم: `admin`
   - كلمة المرور: `lughati1234`

   **غيّري كلمة المرور فوراً من «الإعدادات» بعد أول دخول.**

4) **تشغيل الخادم** (نافذة أوامر):

       npm run dev:server

   تأكدي من: http://localhost:4000/api/health → `{"status":"ok","db":"ok",...}`

5) **تشغيل الواجهة** (نافذة أوامر ثانية):

       npm run dev:web

   ثم افتحي http://localhost:5173

## إدخال الطالبات
من شاشة «الطالبات» ← «استيراد سريع»: سطر لكل طالبة بالصيغة:

    الاسم، الصف، الشعبة

(الصف والشعبة اختياريان — الافتراضي: الرابع / أ). تُتجاهل المكررات تلقائياً.
كذلك يمكن الاستيراد من `database/students.csv` بنفس الصيغة. لا توجد أسماء جاهزة في القاعدة — الإدخال منك فقط.

## الاختبارات الآلية (27 اختباراً ضد PostgreSQL حقيقي)

    npm run db:create -- test
    npm test

تغطي: تسجيل الدخول والجلسات، فرض الصلاحيات في الخادم، خصوصية ملفات الطالبات،
التقييم التراكمي (تقييمان بنفس اليوم)، منع درجات الغائبات، حساب الأسبوع (يبدأ الأحد)،
وإبطال الجلسة عند تغيير كلمة المرور.

## التشغيل الإنتاجي

    npm run build      # يبني الواجهة داخل frontend/dist
    npm start          # يقدم الواجهة والـ API معاً من نفس المنفذ

ملاحظات:
- في `.env` ضعي `NODE_ENV=production` و`JWT_SECRET` قوياً و`CORS_ORIGIN` بنطاقك.
- في الإنتاج تُرسل كوكي الجلسة بعلامة `Secure` — تحتاجين HTTPS (عبر IIS أو Nginx كوكيل عكسي).
- النسخ الاحتياطي الدوري: `pg_dump -U postgres lughati > backup.sql`
- لا ترفعي `.env` ولا مجلد `uploads/` إلى أي مكان عام.

## أمان مدمج
- كلمات المرور مجزأة bcrypt داخل الخادم فقط — لا شيء حساس في كود الواجهة.
- الجلسة JWT داخل كوكي httpOnly (8 ساعات) + `token_version` لإبطال فوري عند تغيير كلمة المرور.
- دور كل مستخدم يُقرأ من قاعدة البيانات مع **كل طلب** — الصلاحيات ليست إخفاء واجهة.
- الطالبة ترى بياناتها فقط؛ ملفاتها لا تُفتح بتغيير الرابط (404 دون كشف).
- سجل تدقيق (audit_logs) للعمليات الحساسة يعرضه دور «الإدارة».
- التقييم تراكمي؛ لا درجات لأي طالبة غير حاضرة (قيد في القاعدة + فحص بالخادم).

ولا تضمّني `.env` عند المشاركة مع أي جهة خارجية.
